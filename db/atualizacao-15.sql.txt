-- ============================================================
--  ATUALIZACAO 15 — PRAZO DA EXECUCAO E NOMES DAS ETAPAS
--
--  1. PRAZO DA EXECUCAO (obra.execucao_prazo)
--     O prazo da OBRA (data_conclusao) e o prazo da ETAPA DE
--     EXECUCAO sao coisas diferentes. A obra pode ter 10 dias e a
--     execucao, 3: depois da execucao ainda ha a documentacao e a
--     entrega. Ate aqui o "Periodo de execucao" gravava o fim dele
--     no prazo final da obra; agora ele tem coluna propria:
--         execucao_inicio -> execucao_prazo   (so a 3a etapa)
--         data_inicio     -> data_conclusao   (a obra inteira)
--
--     A obra que ja tinha usado o periodo antigo (o historico tem um
--     'prazo.execucao' com 'prazo_final') leva aquela data para o
--     prazo da execucao. O prazo final dela fica como esta — o valor
--     de antes nao ficou guardado em lugar nenhum.
--
--  2. NOMES DAS ETAPAS FIXAS
--     1a Comercial, 2a Planejamento, 3a Execucao, 4a Entrega,
--     5a Encerramento. So troca o nome que ainda e o automatico
--     ("1° Etapa", "2a etapa"...): nome escrito por alguem fica.
--
--  SQL simples, sem bloco DO: roda igual no psql, no
--  `npm run db:atualizar` e no editor de SQL do Neon.
--
--  Rodar uma vez. Rodar de novo nao quebra nada e nao apaga nada.
-- ============================================================

BEGIN;

-- 1. o prazo da execucao
ALTER TABLE public.obra ADD COLUMN IF NOT EXISTS execucao_prazo date;

COMMENT ON COLUMN public.obra.execucao_prazo IS
    'Prazo da ETAPA DE EXECUCAO (3a). So ela: a obra continua depois, ate data_conclusao.';

UPDATE public.obra o
   SET execucao_prazo = o.data_conclusao
 WHERE o.execucao_prazo IS NULL
   AND o.data_conclusao IS NOT NULL
   AND EXISTS (SELECT 1 FROM public.atividade a
                WHERE a.acao = 'prazo.execucao'
                  AND a.entidade_id = o.id::text
                  AND a.detalhes ? 'prazo_final');

-- 2. os nomes das etapas fixas
UPDATE public.etapa e
   SET nome = x.nome
  FROM (SELECT f.id, n.nome
          FROM (SELECT id, row_number() OVER (ORDER BY ordem, id) AS posicao
                  FROM public.etapa
                 WHERE fixa AND vigente_ate IS NULL) f
          JOIN (VALUES (1, 'Comercial'), (2, 'Planejamento'), (3, 'Execução'),
                       (4, 'Entrega'), (5, 'Encerramento')) AS n (posicao, nome)
            ON n.posicao = f.posicao) x
 WHERE e.id = x.id
   AND e.nome ~* '^\s*[0-9]+\s*[°ºªo]?\s*etapa\s*$';

COMMIT;

-- ============================================================
--  CONFERENCIA — deve devolver "t" em todas as linhas
-- ============================================================

SELECT 'obra.execucao_prazo' AS item,
       EXISTS (SELECT 1 FROM information_schema.columns
                WHERE table_name = 'obra' AND column_name = 'execucao_prazo') AS ok
UNION ALL
SELECT 'etapas fixas com nome', NOT EXISTS (SELECT 1 FROM public.etapa
                                             WHERE fixa AND vigente_ate IS NULL
                                               AND nome ~* '^\s*[0-9]+\s*[°ºªo]?\s*etapa\s*$');
